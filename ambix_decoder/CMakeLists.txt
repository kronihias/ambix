	SET( PLUGIN_CODE ADe${AMBI_ORDER})
	INCLUDE(${SRC_DIR}/Subproject.cmake)
ENDIF()

# VST3/AU universal
IF(BUILD_VST3 OR BUILD_AU)
	SET(UNIVERSAL_BUILD TRUE)
	SET( PLUGIN_CODE ADeU)
	INCLUDE(${SRC_DIR}/Subproject.cmake)
ENDIF()
